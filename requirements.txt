from reportlab.lib.pagesizes import letter
from reportlab.lib.styles import getSampleStyleSheet, ParagraphStyle
from reportlab.platypus import SimpleDocTemplate, Paragraph, Spacer, Table, TableStyle
from reportlab.lib import colors
from google.colab import files

def generate_pdf():
    pdf_path = "NEURO_STEER_AI_Project_Dossier.pdf"
    doc = SimpleDocTemplate(
        pdf_path,
        pagesize=letter,
        rightMargin=40,
        leftMargin=40,
        topMargin=40,
        bottomMargin=40
    )

    styles = getSampleStyleSheet()
    
    # Custom Typography Styles
    title_style = ParagraphStyle(
        'DocTitle',
        parent=styles['Normal'],
        fontName='Helvetica-Bold',
        fontSize=18,
        leading=22,
        textColor=colors.HexColor('#0B132B')
    )
    sub_style = ParagraphStyle(
        'DocSub',
        parent=styles['Normal'],
        fontName='Helvetica-Bold',
        fontSize=10,
        leading=14,
        textColor=colors.HexColor('#0077B6')
    )
    h1_style = ParagraphStyle(
        'Heading1_Custom',
        parent=styles['Normal'],
        fontName='Helvetica-Bold',
        fontSize=13,
        leading=17,
        spaceBefore=12,
        spaceAfter=6,
        textColor=colors.HexColor('#0B132B')
    )
    body_style = ParagraphStyle(
        'Body_Custom',
        parent=styles['Normal'],
        fontName='Helvetica',
        fontSize=9.5,
        leading=13.5,
        textColor=colors.HexColor('#212529')
    )
    bullet_style = ParagraphStyle(
        'Bullet_Custom',
        parent=body_style,
        leftIndent=15,
        firstLineIndent=-10,
        spaceAfter=4
    )

    story = []

    # Title Banner
    story.append(Paragraph("NEURO-STEER AI: TECHNICAL PROJECT DOSSIER", title_style))
    story.append(Paragraph("Precision Ischemic Stroke Recanalization | Faculty & Mentor Technical Briefing", sub_style))
    story.append(Spacer(1, 10))

    # Section 1
    story.append(Paragraph("1. Project Identity & Metadata", h1_style))
    story.append(Paragraph("• <b>Project Name:</b> NEURO-STEER AI", bullet_style))
    story.append(Paragraph("• <b>Team Identity:</b> Team NEUROX (Neuro-Intervention + Next-Gen Precision)", bullet_style))
    story.append(Paragraph("• <b>Domain:</b> Biomedical Engineering, Micro-Robotics, Cyber-Physical Systems", bullet_style))
    story.append(Paragraph("• <b>Target Pathology:</b> Acute Ischemic Stroke (Distal Microvascular Occlusions)", bullet_style))
    story.append(Spacer(1, 6))

    # Section 2
    story.append(Paragraph("2. Clinical Significance & Gaps in Current Care", h1_style))
    story.append(Paragraph("• <b>Neuronal Attrition:</b> Acute ischemic stroke destroys ~1.9 million neurons per minute ('Time is Brain').", bullet_style))
    story.append(Paragraph("• <b>IV Thrombolysis (tPA) Bottleneck:</b> Narrow 4.5-hour window with a >6% fatal systemic intracranial hemorrhage risk.", bullet_style))
    story.append(Paragraph("• <b>Mechanical Thrombectomy Limit:</b> Catheters cannot safely navigate tortuous micro-vessels (<2 mm) without perforation risk.", bullet_style))
    story.append(Spacer(1, 6))

    # Section 3
    story.append(Paragraph("3. Hardware Architecture & Material Justifications", h1_style))
    
    table_data = [
        [
            Paragraph("<b>Component</b>", body_style),
            Paragraph("<b>Specification</b>", body_style),
            Paragraph("<b>Engineering & Material Justification</b>", body_style)
        ],
        [
            Paragraph("<b>NIR Laser</b>", body_style),
            Paragraph("808 nm Pulsed Diode", body_style),
            Paragraph("Optimal NIR-I biological window: maximizes cranial penetration with zero thermal tissue damage.", body_style)
        ],
        [
            Paragraph("<b>Acoustic Sensor</b>", body_style),
            Paragraph("10-20 MHz Piezoelectric", body_style),
            Paragraph("Detects thermoelastic acoustic expansion; locks 3D clot depth with ±0.2 mm spatial resolution.", body_style)
        ],
        [
            Paragraph("<b>Neural Interface</b>", body_style),
            Paragraph("Multichannel EEG Unit", body_style),
            Paragraph("Monitors Beta/Alpha focus metrics (>70%) to maintain human-in-the-loop safety gating.", body_style)
        ],
        [
            Paragraph("<b>Actuators</b>", body_style),
            Paragraph("4-Coil 3D Array", body_style),
            Paragraph("Generates dynamic 3D magnetic field gradients [F_mag = (m · ∇)B] for spatial vector pulling.", body_style)
        ],
        [
            Paragraph("<b>Nanoparticles</b>", body_style),
            Paragraph("PLGA-coated SPIONs", body_style),
            Paragraph("Superparamagnetic: zero remanence prevents clumping; PLGA ensures safe metabolic clearance.", body_style)
        ]
    ]

    t = Table(table_data, colWidths=[100, 120, 310])
    t.setStyle(TableStyle([
        ('BACKGROUND', (0, 0), (-1, 0), colors.HexColor('#E9ECEF')),
        ('ALIGN', (0, 0), (-1, -1), 'LEFT'),
        ('VALIGN', (0, 0), (-1, -1), 'TOP'),
        ('GRID', (0, 0), (-1, -1), 0.5, colors.HexColor('#CED4DA')),
        ('TOPPADDING', (0, 0), (-1, -1), 4),
        ('BOTTOMPADDING', (0, 0), (-1, -1), 4),
    ]))
    story.append(t)
    story.append(Spacer(1, 8))

    # Section 4
    story.append(Paragraph("4. Software & Control Systems", h1_style))
    story.append(Paragraph("• <b>Digital Twin Interface:</b> Streamlit dashboard backed by WebGL and Plotly 3D hemodynamic modeling.", bullet_style))
    story.append(Paragraph("• <b>Spectral Bandpass Filter:</b> Isolates Beta (13-30 Hz) and Alpha (8-12 Hz) power ratios for cognitive gating.", bullet_style))
    story.append(Paragraph("• <b>AI Co-Pilot Fallback:</b> Reinforcement Learning agent engages when focus drops below 70% to prevent swarm drift.", bullet_style))
    story.append(Paragraph("• <b>System Latency:</b> Hardware-software feedback loop executes within <50 ms latency.", bullet_style))
    story.append(Spacer(1, 6))

    # Section 5
    story.append(Paragraph("5. Quantitative Benchmarks & Phased Roadmap", h1_style))
    story.append(Paragraph("• <b>88.4% Targeting Precision:</b> Validated in simulated Middle Cerebral Artery (MCA) bifurcations.", bullet_style))
    story.append(Paragraph("• <b>>65% Toxicity Reduction:</b> Eliminates non-specific systemic drug dispersal.", bullet_style))
    story.append(Paragraph("• <b>Phase 1 (Completed):</b> In-silico 3D Digital Twin and closed-loop BCI gating validation.", bullet_style))
    story.append(Paragraph("• <b>Phase 2 (Near-Term):</b> Benchtop microfluidic PDMS phantom with Halbach magnetic coil array.", bullet_style))
    story.append(Paragraph("• <b>Phase 3 (Clinical Target):</b> In-vivo animal validation using biodegradable PLGA-SPION carriers.", bullet_style))

    doc.build(story)
    print("PDF generated successfully! Downloading file...")
    files.download(pdf_path)

generate_pdf()
